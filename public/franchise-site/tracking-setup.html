  4. Buat Data Stream → Web → URL: https://garagecoffee.id
  5. Copy Measurement ID (format: G-XXXXXXXXXX)

Microsoft Clarity

// clarity.microsoft.com
  1. Login dengan Microsoft Account (atau buat akun gratis)
  2. "Create new project" → Name: "GARAGE" → Site: garagecoffee.id
  3. Pilih kategori "Marketing"
  4. Copy Project ID (format: alphanumeric pendek)

TikTok Pixel

// ads.tiktok.com
  1. Login ke TikTok Ads Manager (region: Indonesia)
  2. Assets → Events → Web Events → Set Up Web Events
  3. Pilih "Manually install Pixel code"
  4. Beri nama: GARAGE TikTok Pixel
  5. Copy Pixel Code ID (format: alphanumeric)

Universal Tracker Function

Pakai helper ini untuk fire event ke 4 tools sekaligus. Paste sekali di tracking bundle.

// Universal event tracker
window.garageTrack = function(eventName, params = {}) {
  // Google Analytics 4
  if (typeof gtag !== 'undefined') {
    gtag('event', eventName, params);
  }
  // Meta Pixel
  if (typeof fbq !== 'undefined') {
    const standardEvents = ['Lead','Schedule','Contact','ViewContent','InitiateCheckout','CompleteRegistration'];
    if (standardEvents.includes(eventName)) {
      fbq('track', eventName, params);
    } else {
      fbq('trackCustom', eventName, params);
    }
  }
  // TikTok Pixel
  if (typeof ttq !== 'undefined') {
    ttq.track(eventName, params);
  }
  // Console log for debugging
  console.log('[track]', eventName, params);
};

// Usage:
garageTrack('Lead', { value: 250000000, currency: 'IDR', content_name: 'Starter Cafe' });
garageTrack('Schedule', { content_name: 'Konsultasi 30 menit' });
garageTrack('ViewBrosur');  // custom event

Scroll Depth Tracking

Auto-track berapa jauh user scroll. Insight penting: kalau banyak yang bounce di 30%, ada masalah di hero atau load speed.

// Scroll depth tracker (25%, 50%, 75%, 100%)
(function() {
  const milestones = [25, 50, 75, 100];
  const fired = new Set();
  window.addEventListener('scroll', () => {
    const pct = Math.round(
      (window.scrollY / (document.documentElement.scrollHeight - window.innerHeight)) * 100
    );
    milestones.forEach(m => {
      if (pct >= m && !fired.has(m)) {
        fired.add(m);
        if (window.garageTrack) garageTrack('scroll_depth', { depth: m });
      }
    });
  }, { passive: true });
})();

UTM Capture

Capture sumber traffic dari URL UTM params. Pass ke form submission supaya tahu lead dari kampanye mana.

// Capture UTM params, save to sessionStorage, inject into forms
(function() {
  const params = new URLSearchParams(window.location.search);
  const utm = {};
  ['utm_source','utm_medium','utm_campaign','utm_content','utm_term'].forEach(k => {
    const v = params.get(k);
    if (v) utm[k] = v;
  });
  if (Object.keys(utm).length) {
    sessionStorage.setItem('garage_utm', JSON.stringify(utm));
  }
  // Inject into all forms as hidden inputs
  document.addEventListener('submit', (e) => {
    const stored = JSON.parse(sessionStorage.getItem('garage_utm') || '{}');
    Object.entries(stored).forEach(([k, v]) => {
      const input = document.createElement('input');
      input.type = 'hidden';
      input.name = k;
      input.value = v;
      e.target.appendChild(input);
    });
  });
})();
